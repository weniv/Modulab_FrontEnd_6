fetch('https://dev.wenivops.co.kr/services/fastapi-crud/1/signup', {
method: 'POST',
    headers: {
        'Content-Type': 'application/json',
    },
    body: JSON.stringify({
        username: 'hojun1',
        password: 'hojun1234',
    }),
})
.then((response) => response.json())
.then((json) => console.log(json))
.catch((error) => console.error(error));
